## SC1 (→R7) PASS
- 问题: BB 美国 5月上半月各 campaign 的 ROI7、af_cpi、次留,用于放量/观察候选
- 用表: af_cohort_user_acquisition_v2, tj_ad_sdk_revenue, tj_ad_spend_active_v2
- 用 join: ['spend_x_revenue']

```sql
-- block: metrics_tj_ad_spend_active_v2
SELECT campaign_name, SUM(cost_zhe / NULLIF(currency, 0)) / SUM(registers) AS af_cpi
FROM shucang_market.tj_ad_spend_active_v2
WHERE bundle_id = 'com.blockblast.gp' AND media_source = 'Google Ads' AND country = 'US' AND active_date BETWEEN '2026-05-01' AND '2026-05-14'
GROUP BY campaign_name
```

```sql
-- block: roi_7_sdk
WITH spend AS (
  SELECT campaign_name, SUM(cost_zhe / NULLIF(currency,0)) AS cost_zhe
  FROM shucang_market.tj_ad_spend_active_v2 WHERE bundle_id = 'com.blockblast.gp' AND media_source = 'Google Ads' AND country = 'US' AND active_date BETWEEN '2026-05-01' AND '2026-05-14' GROUP BY campaign_name
), rev AS (
  SELECT campaign_name, SUM(revenue) AS revenue_7
  FROM shucang_market.tj_ad_sdk_revenue WHERE bundle_id = 'com.blockblast.gp' AND media_source = 'Google Ads' AND country = 'US' AND active_date BETWEEN '2026-05-01' AND '2026-05-14' AND date_diff <= 6 GROUP BY campaign_name
)
SELECT s.campaign_name, rev.revenue_7 / NULLIF(s.cost_zhe,0) * 100 AS roi_7_sdk
FROM spend s LEFT JOIN rev USING (campaign_name)
```

```sql
-- block: retention

WHERE bundle_id = 'com.blockblast.gp' AND media_source = 'Google Ads' AND active_date BETWEEN '2026-05-01' AND '2026-05-14' AND dt >= '2026-05-01' AND date_diff IN (0, 1)
GROUP BY campaign_name
```
